Q6 - Given two strings text and pattern, count how many substrings of text are anagrams of pattern. An anagram contains exactly the same characters with the same frequencies, although the order may differ. Eg: for, fro, ofr are considered anagram
Input: text = "forxxorfxdofr", pattern = "for"
Output: 3


function subStringsCount(str, pattern){

  let count=0;
  let sorted = pattern.split('').sort().join('');

  for(let i=0;i<str.length;i++){
    let substr = str.substring(i,i+pattern.length);
    let substrSorted = substr.split('').sort().join('');
    if(sorted === substrSorted){
      count++;
    }
  }

  return count;
}
